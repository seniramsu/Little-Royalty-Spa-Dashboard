Install dev dep: `pnpm add -D tsx`. Create `scripts/seed-auth.ts`:
```ts
import { createClient } from "@supabase/supabase-js"
const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
)
const owners = [
  { email: "owner1@littleroyaltyspa.com", password: "ChangeMe-Temp-1!" },
  { email: "owner2@littleroyaltyspa.com", password: "ChangeMe-Temp-2!" },
]
for (const o of owners) {
  const { error } = await supabase.auth.admin.createUser({
    email: o.email, password: o.password, email_confirm: true,
  })
  if (error && !/already/i.test(error.message)) throw error
  console.log("seeded", o.email)
}
```
Add to `package.json` scripts: `"seed:auth": "tsx --env-file=.env.local scripts/seed-auth.ts"`.
